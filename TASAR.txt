         aerror := 1;
         logger.log_error('Error al tasar ' || TO_CHAR(gexpedicion), l_scope);
         ltraza :=
               'Expedicion='
            || TO_CHAR (gexpedicion)
            || ' Ultima linea ejecutada='
            || TO_CHAR (llineaejecutada)
            || ' Error='
            || SQLCODE
            || ' '
            || SQLERRM
            || '#';

         --PMM 09/03/2011
         IF adepurando > 0
         THEN
            INSERT INTO aaa
                        (keyaaa, expedicion, texto, hora
                        )
                 VALUES (aaa_sec.NEXTVAL, gexpedicion, ltraza, SYSDATE
                        );
         END IF;
   END;
